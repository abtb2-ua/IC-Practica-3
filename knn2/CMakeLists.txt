cmake_minimum_required(VERSION 3.15)
project(knn2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Sin tipo de compilación CMake no optimiza (-O0): por defecto Release con -O2,
# el mismo nivel analizado con gprof en la tarea 1. Afecta igual a cv y cv_mpi.
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Tipo de compilación" FORCE)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
add_executable(cv main.cc crossValidation.cc mnist.cc knn.cc sample.cc)
target_include_directories(cv PRIVATE ${PROJECT_SOURCE_DIR})
include(CTest)
if(BUILD_TESTING)
  add_executable(knn2_tests tests.cc sample.cc knn.cc mnist.cc crossValidation.cc)
  add_test(NAME knn2_tests COMMAND knn2_tests)
endif()
find_package(MPI REQUIRED)
add_executable(cv_mpi knn_mpi.cc mnist.cc knn.cc sample.cc)
target_include_directories(cv_mpi PRIVATE ${PROJECT_SOURCE_DIR})
target_link_libraries(cv_mpi MPI::MPI_CXX)
